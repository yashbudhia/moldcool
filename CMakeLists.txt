cmake_minimum_required(VERSION 3.16)
project(moldcool LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(MOLDCOOL_OPENMP "Parallelise stencil and CG loops with OpenMP" ON)
option(MOLDCOOL_WITH_EIGEN "Build the Eigen cross-check test (needs Eigen3)" OFF)
option(MOLDCOOL_CUDA "Build the CUDA explicit-stencil benchmark" OFF)

add_library(moldcool INTERFACE)
target_include_directories(moldcool INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(moldcool INTERFACE /W4 /permissive-)
else()
  target_compile_options(moldcool INTERFACE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  # Link the runtime statically. A MinGW binary that picks up another toolchain's
  # libstdc++-6.dll from PATH (Git for Windows ships one) crashes inside std::ofstream.
  target_link_options(moldcool INTERFACE -static)
endif()

if(MOLDCOOL_OPENMP)
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(moldcool INTERFACE OpenMP::OpenMP_CXX)
    message(STATUS "OpenMP enabled")
  else()
    message(STATUS "OpenMP not found, building serial")
  endif()
endif()

add_executable(moldcool_cli src/moldcool.cpp)
target_link_libraries(moldcool_cli PRIVATE moldcool)
set_target_properties(moldcool_cli PROPERTIES OUTPUT_NAME moldcool)

add_executable(moldcool_bench bench/bench.cpp)
target_link_libraries(moldcool_bench PRIVATE moldcool)

enable_testing()
set(MOLDCOOL_RESULTS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/results)
file(MAKE_DIRECTORY ${MOLDCOOL_RESULTS_DIR})
foreach(t slab_convergence temporal_order stability mms_2d energy_balance cooling_time)
  add_executable(test_${t} tests/test_${t}.cpp)
  target_link_libraries(test_${t} PRIVATE moldcool)
  target_include_directories(test_${t} PRIVATE tests)
  add_test(NAME ${t} COMMAND test_${t} ${MOLDCOOL_RESULTS_DIR})
endforeach()

if(MOLDCOOL_WITH_EIGEN)
  find_package(Eigen3 3.3 QUIET)
  if(NOT Eigen3_FOUND)
    include(FetchContent)
    FetchContent_Declare(eigen
      URL https://gitlab.com/libeigen/eigen/-/archive/3.4.0/eigen-3.4.0.tar.gz
      URL_HASH SHA256=8586084f71f9bde545ee7fa6d00288b264a2b7ac3607b974e54d13e7162c1c72)
    FetchContent_GetProperties(eigen)
    if(NOT eigen_POPULATED)
      FetchContent_Populate(eigen)
    endif()
    add_library(Eigen3::Eigen INTERFACE IMPORTED)
    set_target_properties(Eigen3::Eigen PROPERTIES INTERFACE_INCLUDE_DIRECTORIES ${eigen_SOURCE_DIR})
  endif()
  add_executable(test_cg_vs_eigen tests/test_cg_vs_eigen.cpp)
  target_link_libraries(test_cg_vs_eigen PRIVATE moldcool Eigen3::Eigen)
  target_include_directories(test_cg_vs_eigen PRIVATE tests)
  add_test(NAME cg_vs_eigen COMMAND test_cg_vs_eigen)
endif()

if(MOLDCOOL_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  add_executable(moldcool_cuda_bench cuda/ftcs2d.cu)
  target_link_libraries(moldcool_cuda_bench PRIVATE moldcool)
  set_target_properties(moldcool_cuda_bench PROPERTIES CUDA_ARCHITECTURES "86")
endif()
